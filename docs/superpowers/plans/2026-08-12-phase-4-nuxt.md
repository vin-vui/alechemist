## Contexte pour l'implémenteur (zéro connaissance préalable)

L'application est un outil de suivi de brassage de bière. Modèle de données (schéma DB inchangé depuis le début, aucune migration à créer) :

- `Recipe` (recette, appartient à un `User`) avec des `Step` (ingrédients/actions typés).
- `Brewing` (brassin = exécution d'une recette, snapshot des valeurs de la recette) avec des `BrewingStep` (copie des `Step` au moment de la création, avec un booléen `status` = coché/fait).
- `Brewing.current_step` : enum `preparation|mash|boil|yeast|ferment|completed` (nullable en base). Sert aussi de segment d'URL.
- Timestamps d'étapes sur `Brewing` : `boil_start` (posé par start-boil), `ferment_start` (posé à l'entrée en ferment), `ferment_end` (posé à l'entrée en completed). Il n'existe PAS de `boil_end` : la fin d'ébullition se calcule `boil_start + boil_time` minutes.
- `BrewingStep.type` (enum partagé avec `Step`) : `Preparation`, `Mash`, `Rinse`, `Boil`, `Yeast`, `Primary`, `Secondary`, `Tertiary`, `Bottle`, `Dry Hop`, `Aroma`. Répartition par écran : preparation affiche `Preparation` ; mash affiche `Mash` ; boil affiche `Boil` + `Aroma` ; yeast affiche `Yeast` ; ferment affiche `Primary`,`Secondary`,`Tertiary`,`Bottle`. `Rinse` n'est affiché nulle part.
- `BrewingStep.time` pour les types de boil : temps AVANT LA FIN de l'ébullition (convention BeerXML), pas un temps écoulé. Les `Aroma` s'ajoutent au flame-out (à la fin du chrono). Pour les types ferment : durée en JOURS de l'étape de fermentation.

API existante (phase 3) : 14 routes sous `/api` protégées `auth:sanctum`, mode SPA Sanctum (`statefulApi()` actif dans `bootstrap/app.php`). Fortify headless opérationnel et verrouillé par `tests/Feature/Api/FortifyHeadlessTest.php` : POST `/login` (JSON `{two_factor: bool}`), POST `/logout`, POST `/register`, POST `/two-factor-challenge`, POST `/forgot-password`, POST `/reset-password`, PUT `/user/profile-information`, PUT `/user/password`, POST `/user/confirm-password`, GET `/user/confirmed-password-status`, routes 2FA `/user/two-factor-*`. La vérification d'email est DÉSACTIVÉE (`Features::emailVerification()` commentée) : aucune page de vérification à créer côté Nuxt.

```php
namespace App\Policies;

use App\Models\Recipe;
use App\Models\User;
use Illuminate\Auth\Access\Response;

class RecipePolicy
{
    public function view(User $user, Recipe $recipe): Response
    {

            ? Response::allow()
            : Response::denyAsNotFound();

            ? Response::allow()
            : Response::denyAsNotFound();
    }
}
```

```php
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class UserSessionsController extends Controller
{
    public function index(Request $request)
    {
        if (config('session.driver') !== 'database') {
            return response()->json(['data' => []]);
        }

        $sessions = DB::table('sessions')
            ->where('user_id', $request->user()->id)
            ->orderByDesc('last_activity')
            ->get()
            ->map(fn ($session) => [
                'ip_address' => $session->ip_address,
                'user_agent' => $session->user_agent,
                'last_active_at' => Carbon::createFromTimestamp($session->last_activity)->toJSON(),
                'is_current_device' => $session->id === $request->session()->getId(),
            ])
            ->values();

        return response()->json(['data' => $sessions]);
    }

    public function logoutOthers(Request $request)
    {
        $request->validate(['password' => ['required', 'current_password:web']]);

        Auth::guard('web')->logoutOtherDevices($request->input('password'));

        if (config('session.driver') === 'database') {
            DB::table('sessions')
                ->where('user_id', $request->user()->id)
                ->where('id', '!=', $request->session()->getId())
                ->delete();
        }

        return response()->noContent();
    }
}
```

```php
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Brewing;
use App\Models\BrewingStep;
use App\Models\Recipe;
use App\Models\Step;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class DeleteUserController extends Controller
{
    public function __invoke(Request $request)
    {
        $request->validate(['password' => ['required', 'current_password:web']]);

        $user = $request->user();

        DB::transaction(function () use ($user) {
            $recipeIds = Recipe::where('user_id', $user->id)->pluck('id');
            $brewingIds = Brewing::whereIn('recipe_id', $recipeIds)->pluck('id');

            BrewingStep::whereIn('brewing_id', $brewingIds)->delete();
            Brewing::whereIn('id', $brewingIds)->delete();
            Step::whereIn('recipe_id', $recipeIds)->delete();
            Recipe::whereIn('id', $recipeIds)->delete();

            $user->tokens()->delete();
            $user->delete();
        });

        Auth::guard('web')->logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return response()->noContent();
    }
}
```

`frontend/app/utils/ferment.ts` :

```typescript
import type { BrewingStep } from '~/types/api'

export const FERMENT_TYPES = ['Primary', 'Secondary', 'Tertiary', 'Bottle'] as const

export interface FermentStepView {
  step: BrewingStep
  endsAt: Date
  daysLeft: number
  progress: number
  late: boolean
}

const DAY_MS = 86_400_000

export function fermentSchedule(fermentStart: string, steps: BrewingStep[], now: Date): FermentStepView[] {
  const relevant = steps.filter((s) => (FERMENT_TYPES as readonly string[]).includes(s.type))
  const total = relevant.reduce((sum, s) => sum + (s.time ?? 0), 0)
  let cumulativeDays = 0

  return relevant.map((s) => {
    cumulativeDays += s.time ?? 0
    const endsAt = new Date(new Date(fermentStart).getTime() + cumulativeDays * DAY_MS)
    const msLeft = endsAt.getTime() - now.getTime()
    const daysLeftRaw = msLeft / DAY_MS
    const progress = total > 0 ? Math.min(100, Math.max(0, (1 - daysLeftRaw / total) * 100)) : 100

    return {
      step: s,
      endsAt,
      daysLeft: Math.max(0, Math.ceil(daysLeftRaw)),
      progress,
      late: msLeft < 0 && !s.status,
    }
  })
}
```

```typescript
// Echelle de couleur EBC transcrite de resources/css/app.css (.ebc-N).
const EBC_HEX: Record<number, string> = {
  // 0: '#......', 1: '#......', ... recopier chaque paire N -> hex du fichier source
}

const MIN_EBC = Math.min(...Object.keys(EBC_HEX).map(Number))
const MAX_EBC = Math.max(...Object.keys(EBC_HEX).map(Number))

export function ebcToHex(ebc: number | string | null): string {
  const n = Math.round(Number(ebc ?? MIN_EBC))
  const clamped = Math.min(MAX_EBC, Math.max(MIN_EBC, Number.isFinite(n) ? n : MIN_EBC))
  // Si un N intermediaire manque dans l'echelle source, prendre le plus proche en dessous.
  for (let i = clamped; i >= MIN_EBC; i--) {
    if (EBC_HEX[i]) return EBC_HEX[i]
  }
  return EBC_HEX[MIN_EBC]
}
```

```typescript
import type { BrewingStep, CurrentStep, StepType } from '~/types/api'

export interface StepConfig {
  types: StepType[]
  next: CurrentStep
  title: string
}

export type StepSlug = Exclude<CurrentStep, 'completed'>

export const STEP_CONFIG: Record<StepSlug, StepConfig> = {
  preparation: { types: ['Preparation'], next: 'mash', title: 'Preparation' },
  mash: { types: ['Mash'], next: 'boil', title: 'Mash' },
  boil: { types: ['Boil', 'Aroma'], next: 'yeast', title: 'Boil' },
  yeast: { types: ['Yeast'], next: 'ferment', title: 'Yeast' },
  ferment: { types: ['Primary', 'Secondary', 'Tertiary', 'Bottle'], next: 'completed', title: 'Ferment' },
}

export const STEP_ORDER: CurrentStep[] = ['preparation', 'mash', 'boil', 'yeast', 'ferment', 'completed']

export function isStepSlug(value: unknown): value is StepSlug {
  return typeof value === 'string' && value in STEP_CONFIG
}

export function stepsForPage(stepKey: StepSlug, steps: BrewingStep[]): BrewingStep[] {
  const filtered = steps.filter((s) => STEP_CONFIG[stepKey].types.includes(s.type))
  if (stepKey === 'preparation') {
    // Parite web (orderBy unit DESC) : les volumes a chauffer d'abord.
    return [...filtered.filter((s) => s.unit === 'l'), ...filtered.filter((s) => s.unit !== 'l')]
  }
  return filtered
}
```

```typescript
import type { BrewingStep } from '~/types/api'

const MIN_MS = 60_000

export function boilEndMs(boilStart: string | null, boilTimeMin: number): number | null {
  if (!boilStart) return null
  return Date.parse(boilStart) + boilTimeMin * MIN_MS
}

export function boilRemainingMs(boilStart: string | null, boilTimeMin: number, nowMs: number): number | null {
  const end = boilEndMs(boilStart, boilTimeMin)
  return end === null ? null : end - nowMs
}

export function stepRemainingMinutes(
  boilStart: string | null,
  boilTimeMin: number,
  step: BrewingStep,
  nowMs: number,
): number | null {
  const end = boilEndMs(boilStart, boilTimeMin)
  if (end === null) return null
  const target = step.type === 'Aroma' ? end : end - (step.time ?? 0) * MIN_MS
  return Math.ceil((target - nowMs) / MIN_MS)
}

export function isStepLate(
  boilStart: string | null,
  boilTimeMin: number,
  step: BrewingStep,
  nowMs: number,
): boolean {
  const remaining = stepRemainingMinutes(boilStart, boilTimeMin, step, nowMs)
  return remaining !== null && remaining <= 0 && !step.status
}
```

```typescript
export function useConfirmPassword() {
  const client = useSanctumClient()
  const modalVisible = ref(false)
  const modalError = ref('')
  let pendingAction: (() => Promise<void>) | null = null

  async function runConfirmed(action: () => Promise<void>) {
    try {
      await action()
    } catch (e) {
      if (!needsPasswordConfirmation(e)) throw e
      pendingAction = action
      modalVisible.value = true
    }
  }

  async function confirm(password: string) {
    modalError.value = ''
    try {
      await client('/user/confirm-password', { method: 'POST', body: { password } })
      modalVisible.value = false
      if (pendingAction) {
        const action = pendingAction
        pendingAction = null
        await action()
      }
    } catch (e) {
      const errors = extractErrors(e)
      modalError.value = errors.password?.[0] ?? 'Erreur'
    }
  }

  return { modalVisible, modalError, runConfirmed, confirm }
}
```
